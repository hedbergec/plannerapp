    comparePicka_choices <- reactive({
    req(input$preAssignedTreatment == "noSet")
    req(qedData())
    req(qedData_stats())
    req(input$treatpickeda)
    return(
      qedData_stats()$total$school[!(qedData_stats()$total$school %in% input$treatpickeda)]
    )
  })

    req(input$preAssignedTreatment == "noSet")

    selectizeInput("comparepickeda",

                   multiple = TRUE)
